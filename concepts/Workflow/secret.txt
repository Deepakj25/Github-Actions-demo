//Explicit secret
Reusable workflow:
on:
  workflow_call:
    secrets:
      AZURE_CLIENT_SECRET:
        required: true

Caller:

jobs:
  deploy:
    uses: ./.github/workflows/deploy.yml
    secrets:
      AZURE_CLIENT_SECRET: ${{ secrets.AZURE_CLIENT_SECRET }}

//this is a simple ....resuable workflow ask for a secret....caller sends the secret


// inherit
jobs:
  deploy:
    uses: ./.github/workflows/deploy.yml
    secrets: inherit

reusable workflow  has access to the caller's available secrets rather than explicitly documenting a narrow interface.

can be used as ${{ secrets.AZURE_CLIENT_SECRET }} in caller.
